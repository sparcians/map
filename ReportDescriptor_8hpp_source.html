431 std::string stringize() const {
432 auto comma_sep_files = [](const std::vector<std::string>& dest_files)
433 {
434 std::ostringstream oss;
435 bool comma = false;
436 for (const auto& s : dest_files)
437 {
438 oss << s;
439 if (comma)
440 {
441 oss << ", ";
442 }
443 comma = true;
444 }
445 return oss.str();
446 };
447
448 auto _dest_files = sprintf_notif_trigger_dest_files_.empty() ?
449 (orig_dest_file_.empty() ? dest_file : orig_dest_file_) :
450 comma_sep_files(sprintf_notif_trigger_dest_files_);
451
452 std::stringstream ss;
453 ss << "Report def \"" << def_file << "\" on node \"" << loc_pattern
454 << "\" -> \"" << _dest_files << "\"";
455 if(format.size() != 0){
456 ss << " (format=" << format << ")";
457 }
458 return ss.str();
459 }

677 void push_back(const ReportDescriptor & rd) {
678 //Internal book-keeping is only for Python shell workflows.
679 //We also do not perform any book-keeping for descriptors
680 //whose dest_file is "1" (print to stdout). That is the
681 //only use case where duplicate dest_file's is actually
682 //okay. We will add the descriptor to the collection, but
683 //any attempts to access this descriptor later (i.e. to
684 //change its parameters or to disable it from Python) will
685 //throw an exception that the descriptor "1" cannot be found.
686 if (rd.dest_file != "1") {
687 const std::string desc_name = getDescriptorName_(rd);
688 auto iter = indices_by_descriptor_name_.find(desc_name);
689 if (iter != indices_by_descriptor_name_.end()) {
690 //Descriptor found by this name. Throw if it is already
691 //enabled. Users can only replace disabled descriptors.
692 if (rep_descs_.at(iter->second).isEnabled()) {
693 throw SpartaException("Report descriptor named '") <<
694 desc_name << "' already exists in this configuration";
695 }
696 }
697 indices_by_descriptor_name_[desc_name] = rep_descs_.size();
698 }
699
700 //This incoming 'rd' variable could be a reference to a
701 //previously disabled descriptor:
702 //
703 // >>> rd = report_config.descriptors.foo_csv
704 // >>> report_config.removeReport('foo_csv')
705 // >>> report_config.addReport(rd)
706 //
707 //This would still count as an enabled descriptor.
708 rep_descs_.push_back(rd);
709 rep_descs_.back().enabled_ = true;
710 }

744 bool contains(const std::string & desc_name) const {
745 auto iter = indices_by_descriptor_name_.find(desc_name);
746 if (iter == indices_by_descriptor_name_.end()) {
747 //The name may have been given to us with a period, not
748 //an underscore, separating the file stem and the extension
749 const auto replaced_desc_name = replaceDotsWithUnderscores_(desc_name);
750 iter = indices_by_descriptor_name_.find(replaced_desc_name);
751 }
752
753 //If this name is not even found in our map, we do not
754 //contain this descriptor
755 if (iter == indices_by_descriptor_name_.end()) {
756 return false;
757 }
758
759 //But even if the descriptor is physcially here, we have
760 //to also ask it if is is still enabled
761 return rep_descs_.at(iter->second).isEnabled();
762 }

819 size_t getDescriptorIndexByName_(const std::string & desc_name) const {
820 auto iter = indices_by_descriptor_name_.find(desc_name);
821 if (iter == indices_by_descriptor_name_.end()) {
822 //The name may have been given to us with a period, not
823 //an underscore, separating the file stem and the extension
824 const auto replaced_desc_name = replaceDotsWithUnderscores_(desc_name);
825 iter = indices_by_descriptor_name_.find(replaced_desc_name);
826
827 //If it still is not found, error out
828 if (iter == indices_by_descriptor_name_.end()) {
829 throw SpartaException(
830 "No descriptor named '") << desc_name << "' exists";
831 }
832 }
833
834 //The descriptor exists in this collection, but we still need
835 //to throw if it has been disabled, since we are not supposed
836 //to be using those descriptors for any reason anymore.
837 if (!rep_descs_.at(iter->second).isEnabled()) {
838 throw SpartaException("The descriptor named '") << desc_name
839 << "' has already been disabled";
840 }
841 return iter->second;
842 }
