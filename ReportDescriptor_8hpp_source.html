417 std::string stringize() const {
418 auto comma_sep_files = [](const std::vector<std::string>& dest_files)
419 {
420 std::ostringstream oss;
421 bool comma = false;
422 for (const auto& s : dest_files)
423 {
424 oss << s;
425 if (comma)
426 {
427 oss << ", ";
428 }
429 comma = true;
430 }
431 return oss.str();
432 };
433
434 auto _dest_files = sprintf_notif_trigger_dest_files_.empty() ?
435 (orig_dest_file_.empty() ? dest_file : orig_dest_file_) :
436 comma_sep_files(sprintf_notif_trigger_dest_files_);
437
438 std::stringstream ss;
439 ss << "Report def \"" << def_file << "\" on node \"" << loc_pattern
440 << "\" -> \"" << _dest_files << "\"";
441 if(format.size() != 0){
442 ss << " (format=" << format << ")";
443 }
444 return ss.str();

663 void push_back(const ReportDescriptor & rd) {
664 //Internal book-keeping is only for Python shell workflows.
665 //We also do not perform any book-keeping for descriptors
666 //whose dest_file is "1" (print to stdout). That is the
667 //only use case where duplicate dest_file's is actually
668 //okay. We will add the descriptor to the collection, but
669 //any attempts to access this descriptor later (i.e. to
670 //change its parameters or to disable it from Python) will
671 //throw an exception that the descriptor "1" cannot be found.
672 if (rd.dest_file != "1") {
673 const std::string desc_name = getDescriptorName_(rd);
674 auto iter = indices_by_descriptor_name_.find(desc_name);
675 if (iter != indices_by_descriptor_name_.end()) {
676 //Descriptor found by this name. Throw if it is already
677 //enabled. Users can only replace disabled descriptors.
678 if (rep_descs_.at(iter->second).isEnabled()) {
679 throw SpartaException("Report descriptor named '") <<
680 desc_name << "' already exists in this configuration";
681 }
682 }
683 indices_by_descriptor_name_[desc_name] = rep_descs_.size();
684 }
685
686 //This incoming 'rd' variable could be a reference to a
687 //previously disabled descriptor:
688 //
689 // >>> rd = report_config.descriptors.foo_csv
690 // >>> report_config.removeReport('foo_csv')
691 // >>> report_config.addReport(rd)
692 //
693 //This would still count as an enabled descriptor.
694 rep_descs_.push_back(rd);
695 rep_descs_.back().enabled_ = true;
696 }

730 bool contains(const std::string & desc_name) const {
731 auto iter = indices_by_descriptor_name_.find(desc_name);
732 if (iter == indices_by_descriptor_name_.end()) {
733 //The name may have been given to us with a period, not
734 //an underscore, separating the file stem and the extension
735 const auto replaced_desc_name = replaceDotsWithUnderscores_(desc_name);
736 iter = indices_by_descriptor_name_.find(replaced_desc_name);
737 }
738
739 //If this name is not even found in our map, we do not
740 //contain this descriptor
741 if (iter == indices_by_descriptor_name_.end()) {
742 return false;
743 }
744
745 //But even if the descriptor is physcially here, we have
746 //to also ask it if is is still enabled
747 return rep_descs_.at(iter->second).isEnabled();
748 }

805 size_t getDescriptorIndexByName_(const std::string & desc_name) const {
806 auto iter = indices_by_descriptor_name_.find(desc_name);
807 if (iter == indices_by_descriptor_name_.end()) {
808 //The name may have been given to us with a period, not
809 //an underscore, separating the file stem and the extension
810 const auto replaced_desc_name = replaceDotsWithUnderscores_(desc_name);
811 iter = indices_by_descriptor_name_.find(replaced_desc_name);
812
813 //If it still is not found, error out
814 if (iter == indices_by_descriptor_name_.end()) {
815 throw SpartaException(
816 "No descriptor named '") << desc_name << "' exists";
817 }
818 }
819
820 //The descriptor exists in this collection, but we still need
821 //to throw if it has been disabled, since we are not supposed
822 //to be using those descriptors for any reason anymore.
823 if (!rep_descs_.at(iter->second).isEnabled()) {
824 throw SpartaException("The descriptor named '") << desc_name
825 << "' has already been disabled";
826 }
827 return iter->second;
828 }
