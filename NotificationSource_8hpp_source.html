351 static void validateNotificationName(const std::string& nm)
352 {
353 for(const char* rsrv : RESERVED_WORDS){
354 if(nm == rsrv){
355 throw SpartaException("Notification name \"")
356 << nm << "\" is a reserved word. ";
357 }
358 }
359
360 if(nm.size() > 0 && nm[0] == '_'){
361 throw SpartaException(" Notification name \"")
362 << nm << "\" begins with an '_' which is not permitted";
363 }
364
365 if(nm.find("__") != std::string::npos){
366 throw SpartaException(" Notification name \"")
367 << nm << "\" contains two adjacent underscores which is not permitted";
368 }
369
370 if(nm.find_first_of(DIGIT_CHARS) == 0){
371 throw SpartaException(" Notification name \"")
372 << nm << "\" begins with a '" << nm[nm.size()-1] << "' character which is not permitted. "
373 << "A Notification name must not begin with a decimal digit.";
374 }
375
376 size_t pos = nm.find_first_not_of(ALPHANUM_CHARS);
377 if(pos != std::string::npos){
378 throw SpartaException("Notification name \"")
379 << nm << "\" contains a '" << nm[pos] << "', which is not permitted. "
380 << "A Notification name must contain only alphanumeric characters and underscores.";
381 }

500 // Override from TreeNode
502 virtual void notificationObserverAdded_(const std::type_info& tinfo,
503 const std::string* name_id,
504 TreeNode* obs_node,
505 const delegate* del) override {
506 // Expected to only see observation on the notification that this source can
507 // generate. If these assertions are hit, then the framework is broken or
508 // canGenerateNotification_ is incorrect
509 sparta_assert(tinfo == noti_tinfo_);
510 sparta_assert(name_id == noti_id_ || name_id == StringManager::getStringManager().EMPTY);
511
512 // Add the observation point node if unique
513 auto itr = std::find(obs_nodes_.begin(), obs_nodes_.end(), obs_node);
514 if(itr == obs_nodes_.end()){
515 obs_nodes_.push_back(obs_node);
516 }
517
518 dels_.push_back(*del);
519
520 const bool was_observed = observed_;
521 observed_ = true;
522
523 // Callback happen after state changes are complete (or before) in order to allow recursion
524 if(not was_observed){
525 invokeObservationStateChangeCallbacks_(ObservationStateChange::SOLE_OBSERVER_REGISTERING);
526 }
527 invokeObservationStateChangeCallbacks_(ObservationStateChange::OBSERVER_REGISTERING);
528 }

530 // Override from TreeNode
532 virtual void notificationObserverRemoved_(const std::type_info& tinfo,
533 const std::string* name_id,
534 TreeNode* obs_node,
535 const delegate* del) override {
536 // Expected to only see observation on the notification that this source can
537 // generate. If these assertions are hit, then the framework is broken or
538 // canGenerateNotification_ is incorrect
539 sparta_assert(tinfo == noti_tinfo_);
540 sparta_assert(name_id == noti_id_ || name_id == StringManager::getStringManager().EMPTY);
541
542 const bool was_observed = observed_;
543
544 auto delitr = std::find(dels_.begin(), dels_.end(), *del);
545 if(delitr != dels_.end()){
546 dels_.erase(delitr);
547 }
548
549 observed_ = dels_.size() > 0;
550
551 // Remove the observation point node if no delegates refer to it any more
552 bool remaining = false;
553 for(auto d : dels_){
554 if(d.getObservationPoint() == obs_node){
555 remaining = true;
556 break;
557 }
558 }
559 if(false == remaining){
560 auto itr = std::find(obs_nodes_.begin(), obs_nodes_.end(), obs_node);
561 if(itr != obs_nodes_.end()){
562 obs_nodes_.erase(itr);
563 }
564 }
565
566 // Callback happen after state changes are complete (or before) in order to allow recursion
567 // Reverse order of callback from notificationObserverAdded_.
568 invokeObservationStateChangeCallbacks_(ObservationStateChange::OBSERVER_DEREGISTERING);
569 if(was_observed && not observed_){
570 invokeObservationStateChangeCallbacks_(ObservationStateChange::SOLE_OBSERVER_DEREGISTERING);
571 }
572 }

618 void determineObs_Nodes_() {
619 obs_nodes_.clear();
620 dels_.clear();
621 observed_ = false;
622
623 TreeNode* node = this;
624 while(node){
625 node->getDelegatesRegisteredForNotification(noti_tinfo_, noti_id_, dels_);
626 if(node->hasObserversRegisteredForNotification(noti_tinfo_, noti_id_)){
627 obs_nodes_.push_back(node);

634 if(node->hasObserversRegisteredForNotification(noti_tinfo_, noti_id_)){

651 void registerObservationStateChangeCallback_(ObservationStateCallback* const hook) {
652 // User errors
653 sparta_assert(in_observation_state_change_callback_ == false,
654 "cannot add observation state change callbacks from within a callback");
655
656 // Internal errors
657 sparta_assert(hook); // Internal-only hook argument is null somehow?
658 sparta_assert(std::find(obs_state_change_cbs_.begin(), obs_state_change_cbs_.end(), hook) == obs_state_change_cbs_.end(),
659 "Internal notification hook " << *hook
660 << " was installed a second time to notification source " << *this);
661
662 // Track it in a list
663 obs_state_change_cbs_.emplace_back(hook);
664 }
665
669 void deregisterObservationStateChangeCallback_(ObservationStateCallback* const hook) {
670 // User errors
671 sparta_assert(in_observation_state_change_callback_ == false,
672 "cannot remove observation state change callbacks from within a callback");
673
674 const auto itr = std::find(obs_state_change_cbs_.begin(), obs_state_change_cbs_.end(), hook);
675
676 // Internal error checking
677 sparta_assert(itr != obs_state_change_cbs_.end(),
678 "Internal notification hook " << *hook
679 << " was not found while attemptint to deregister it from notification source " << *this);
680
681 // Remove it so it is no longer called
682 obs_state_change_cbs_.erase(itr);
683 }
684
690 void invokeObservationStateChangeCallbacks_(ObservationStateChange to_call) {
691 // User Error
692 sparta_assert(in_observation_state_change_callback_ == false,
693 "cannot recursively invoke observation state change callbacks from within a "
694 "callback. User may be changing Notification Observers in response to "
695 "changes in observation state. This is not allowed");
696
697 in_observation_state_change_callback_ = true;
698
699 // Break on first exception!
700 try {
701 for(ObservationStateCallback const * cb : obs_state_change_cbs_){
702 if(cb->getType() == to_call){
703 (*cb)(*this, getNumObservers());
704 }
705 }
706 } catch (...) {
707 in_observation_state_change_callback_ = false;
708
709 throw; // Re-throw
710 }
711
712 in_observation_state_change_callback_ = false;
713 }

1005 void setFormatString(const std::string& format_str)
1006 {
1007 // The format_str looks something like "%s_$i.json".
1008 // In the event that we have multiple reports setup
1009 // to be triggered by the same notif source, we need
1010 // to ensure that the format specifiers are the same
1011 // for all ReportDescriptor dest_file's we are given.
1012 auto specs = parseFormatSpecifiers_(format_str);
1013 if (!parsed_specs_.empty())
1014 {
1015 sparta_assert(specs == parsed_specs_);
1016 }
1017 else
1018 {
1019 parsed_specs_ = specs;
1020 }
1021
1022 format_str_ = format_str;
1023 }
1024
1025 bool hasFormatString() const {
1026 return !format_str_.empty();
1027 }

1036 template <typename... Args>
1037 void postNotification(Args... args) const
1038 {
1039 sparta_assert(!format_str_.empty(),
1040 "setFormatString() must be called before postNotification() on " << getLocation());
1041
1042 constexpr std::array<SpecType, sizeof...(Args)> expected_types { specTypeFor_<Args>()... };
1043
1044 if(parsed_specs_.size() != sizeof...(Args)){
1045 throw SpartaException("Sprintf format string \"")
1046 << format_str_ << "\" for " << getLocation() << " has " << parsed_specs_.size()
1047 << " specifier(s) but " << sizeof...(Args) << " argument(s) were given";
1048 }
1049
1050 for(std::size_t i = 0; i < parsed_specs_.size(); ++i){
1051 if(parsed_specs_[i] != expected_types[i]){
1052 throw SpartaException("Sprintf format string \"")
1053 << format_str_ << "\" for " << getLocation()
1054 << " has a type mismatch at specifier index " << i;
1055 }
1056 }
1057
1058 // One stream-to-ostringstream lambda per argument, invoked in order as specifiers are encountered
1059 const std::array<std::function<void(std::ostringstream&)>, sizeof...(Args)> streamers {
1060 [&args](std::ostringstream& o){ o << args; }...
1061 };
1062
1063 std::ostringstream oss;
1064 std::size_t arg_idx = 0;
1065 for(std::size_t i = 0; i < format_str_.size(); ++i){
1066 if(format_str_[i] != '%'){
1067 oss << format_str_[i];
1068 continue;
1069 }
1070
1071 const char spec = format_str_[++i];
1072 if(spec == '%'){
1073 oss << '%';
1074 }else{
1075 streamers[arg_idx++](oss);
1076 }
1077 }
1078
1079 NotificationSource<std::string>::postNotification(oss.str());
1080 }
1081
1082 private:
1083 enum class SpecType { STRING, INT };
1084
1085 // Maps an argument type to the specifier it must match (%i for non-bool integrals, %s otherwise)
1086 template <typename T>
1087 static constexpr SpecType specTypeFor_()
1088 {
1089 using PlainT = typename std::remove_cv<typename std::remove_reference<T>::type>::type;
1090 if constexpr (std::is_integral<PlainT>::value && !std::is_same<PlainT, bool>::value){
1091 return SpecType::INT;
1092 }else{
1093 return SpecType::STRING;
1094 }
1095 }
1096
1097 // Runtime scan of format_str for %s/%i/%% specifiers, in order of appearance
1098 static std::vector<SpecType> parseFormatSpecifiers_(const std::string& format_str)
1099 {
1100 std::vector<SpecType> specs;
1101 for(std::size_t i = 0; i < format_str.size(); ++i){
1102 if(format_str[i] != '%'){
1103 continue;
1104 }
1105
1106 if(i + 1 >= format_str.size()){
1107 throw SpartaException("Sprintf format string \"")
1108 << format_str << "\" ends with a lone '%' character";
1109 }
1110
1111 const char spec = format_str[++i];
1112 switch(spec){
1113 case 's':
1114 specs.push_back(SpecType::STRING);
1115 break;
1116 case 'i':
1117 specs.push_back(SpecType::INT);
1118 break;
1119 case '%':
1120 break; // Literal '%', not a specifier
1121 default:
1122 throw SpartaException("Sprintf format string \"")
1123 << format_str << "\" contains unsupported specifier '%" << spec
1124 << "'. Only %s and %i are supported";
1125 }
1126 }
1127 return specs;
1128 }

1131 std::vector<SpecType> parsed_specs_;

1267 void findMatchingSourcesBelowNode_(sparta::TreeNode& node,
1268 SrcList& bases)
1269 {
1270 auto* as_src = dynamic_cast<NotificationSourceBase*>(&node);
1271 if (as_src)
1272 {
1273 // If this MirrorNotificationSource happened to live under the search
1274 // of node, then we would certainly not want to add ourselves as a
1275 // concrete instance!
1276 if (as_src != this)
1277 {
1278 // We won't mirror anything that isn't explicitly allowed by the concrete
1279 // source.
1280 if (as_src->getNotificationID() == getNotificationID() &&
1281 as_src->getNotificationType() == getNotificationType())
1282 {
1283 bases.emplace_back(dynamic_cast<NotificationSourceBase*>(&node));
1284 }
1285 }
1286 }
