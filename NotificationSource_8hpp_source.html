348 static void validateNotificationName(const std::string& nm)
349 {
350 for(const char* rsrv : RESERVED_WORDS){
351 if(nm == rsrv){
352 throw SpartaException("Notification name \"")
353 << nm << "\" is a reserved word. ";
354 }
355 }
356
357 if(nm.size() > 0 && nm[0] == '_'){
358 throw SpartaException(" Notification name \"")
359 << nm << "\" begins with an '_' which is not permitted";
360 }
361
362 if(nm.find("__") != std::string::npos){
363 throw SpartaException(" Notification name \"")
364 << nm << "\" contains two adjacent underscores which is not permitted";
365 }
366
367 if(nm.find_first_of(DIGIT_CHARS) == 0){
368 throw SpartaException(" Notification name \"")
369 << nm << "\" begins with a '" << nm[nm.size()-1] << "' character which is not permitted. "
370 << "A Notification name must not begin with a decimal digit.";
371 }
372
373 size_t pos = nm.find_first_not_of(ALPHANUM_CHARS);
374 if(pos != std::string::npos){
375 throw SpartaException("Notification name \"")
376 << nm << "\" contains a '" << nm[pos] << "', which is not permitted. "
377 << "A Notification name must contain only alphanumeric characters and underscores.";
378 }

497 // Override from TreeNode
499 virtual void notificationObserverAdded_(const std::type_info& tinfo,
500 const std::string* name_id,
501 TreeNode* obs_node,
502 const delegate* del) override {
503 // Expected to only see observation on the notification that this source can
504 // generate. If these assertions are hit, then the framework is broken or
505 // canGenerateNotification_ is incorrect
506 sparta_assert(tinfo == noti_tinfo_);
507 sparta_assert(name_id == noti_id_ || name_id == StringManager::getStringManager().EMPTY);
508
509 // Add the observation point node if unique
510 auto itr = std::find(obs_nodes_.begin(), obs_nodes_.end(), obs_node);
511 if(itr == obs_nodes_.end()){
512 obs_nodes_.push_back(obs_node);
513 }
514
515 dels_.push_back(*del);
516
517 const bool was_observed = observed_;
518 observed_ = true;
519
520 // Callback happen after state changes are complete (or before) in order to allow recursion
521 if(not was_observed){
522 invokeObservationStateChangeCallbacks_(ObservationStateChange::SOLE_OBSERVER_REGISTERING);
523 }
524 invokeObservationStateChangeCallbacks_(ObservationStateChange::OBSERVER_REGISTERING);
525 }
526
527 // Override from TreeNode
529 virtual void notificationObserverRemoved_(const std::type_info& tinfo,
530 const std::string* name_id,
531 TreeNode* obs_node,
532 const delegate* del) override {
533 // Expected to only see observation on the notification that this source can
534 // generate. If these assertions are hit, then the framework is broken or
535 // canGenerateNotification_ is incorrect
536 sparta_assert(tinfo == noti_tinfo_);
537 sparta_assert(name_id == noti_id_ || name_id == StringManager::getStringManager().EMPTY);
538
539 const bool was_observed = observed_;
540
541 auto delitr = std::find(dels_.begin(), dels_.end(), *del);
542 if(delitr != dels_.end()){
543 dels_.erase(delitr);
544 }
545
546 observed_ = dels_.size() > 0;
547
548 // Remove the observation point node if no delegates refer to it any more
549 bool remaining = false;
550 for(auto d : dels_){
551 if(d.getObservationPoint() == obs_node){
552 remaining = true;
553 break;
554 }
555 }
556 if(false == remaining){
557 auto itr = std::find(obs_nodes_.begin(), obs_nodes_.end(), obs_node);
558 if(itr != obs_nodes_.end()){
559 obs_nodes_.erase(itr);
560 }
561 }
562
563 // Callback happen after state changes are complete (or before) in order to allow recursion
564 // Reverse order of callback from notificationObserverAdded_.
565 invokeObservationStateChangeCallbacks_(ObservationStateChange::OBSERVER_DEREGISTERING);
566 if(was_observed && not observed_){
567 invokeObservationStateChangeCallbacks_(ObservationStateChange::SOLE_OBSERVER_DEREGISTERING);
568 }
569 }

615 void determineObs_Nodes_() {
616 obs_nodes_.clear();
617 dels_.clear();
618 observed_ = false;
619
620 TreeNode* node = this;
621 while(node){
622 node->getDelegatesRegisteredForNotification(noti_tinfo_, noti_id_, dels_);
623 if(node->hasObserversRegisteredForNotification(noti_tinfo_, noti_id_)){
624 obs_nodes_.push_back(node);
625 }
626 node = node->getParent();
627 }
628
629 node = getVirtualGlobalNode();
630 node->getDelegatesRegisteredForNotification(noti_tinfo_, noti_id_, dels_);
631 if(node->hasObserversRegisteredForNotification(noti_tinfo_, noti_id_)){
632 obs_nodes_.push_back(node);
633 }
634
635 // Update observed flag
636 observed_ = dels_.size() > 0;
637 }

648 void registerObservationStateChangeCallback_(ObservationStateCallback* const hook) {
649 // User errors
650 sparta_assert(in_observation_state_change_callback_ == false,
651 "cannot add observation state change callbacks from within a callback");
652
653 // Internal errors
654 sparta_assert(hook); // Internal-only hook argument is null somehow?
655 sparta_assert(std::find(obs_state_change_cbs_.begin(), obs_state_change_cbs_.end(), hook) == obs_state_change_cbs_.end(),
656 "Internal notification hook " << *hook
657 << " was installed a second time to notification source " << *this);
658
659 // Track it in a list
660 obs_state_change_cbs_.emplace_back(hook);
661 }
662
666 void deregisterObservationStateChangeCallback_(ObservationStateCallback* const hook) {
667 // User errors
668 sparta_assert(in_observation_state_change_callback_ == false,
669 "cannot remove observation state change callbacks from within a callback");
670
671 const auto itr = std::find(obs_state_change_cbs_.begin(), obs_state_change_cbs_.end(), hook);
672
673 // Internal error checking
674 sparta_assert(itr != obs_state_change_cbs_.end(),
675 "Internal notification hook " << *hook
676 << " was not found while attemptint to deregister it from notification source " << *this);
677
678 // Remove it so it is no longer called
679 obs_state_change_cbs_.erase(itr);
680 }
681
687 void invokeObservationStateChangeCallbacks_(ObservationStateChange to_call) {
688 // User Error
689 sparta_assert(in_observation_state_change_callback_ == false,
690 "cannot recursively invoke observation state change callbacks from within a "
691 "callback. User may be changing Notification Observers in response to "
692 "changes in observation state. This is not allowed");
693
694 in_observation_state_change_callback_ = true;
695
696 // Break on first exception!
697 try {
698 for(ObservationStateCallback const * cb : obs_state_change_cbs_){
699 if(cb->getType() == to_call){
700 (*cb)(*this, getNumObservers());
701 }
702 }
703 } catch (...) {
704 in_observation_state_change_callback_ = false;
705
706 throw; // Re-throw
707 }
708
709 in_observation_state_change_callback_ = false;
710 }
711
720 std::vector<ObservationStateCallback*> obs_state_change_cbs_;
721
730 bool in_observation_state_change_callback_ = false;

1002 void setFormatString(const std::string& format_str)
1003 {
1004 // The format_str looks something like "%s_$i.json".
1005 // In the event that we have multiple reports setup
1006 // to be triggered by the same notif source, we need
1007 // to ensure that the format specifiers are the same
1008 // for all ReportDescriptor dest_file's we are given.
1009 auto specs = parseFormatSpecifiers_(format_str);
1010 if (!parsed_specs_.empty())
1011 {
1012 sparta_assert(specs == parsed_specs_);
1013 }
1014 else
1015 {
1016 parsed_specs_ = specs;
1017 }
1018
1019 format_str_ = format_str;
1020 }
1021
1022 bool hasFormatString() const {
1023 return !format_str_.empty();
1024 }

1033 template <typename... Args>
1034 void postNotification(Args... args) const
1035 {
1036 sparta_assert(!format_str_.empty(),
1037 "setFormatString() must be called before postNotification() on " << getLocation());
1038
1039 constexpr std::array<SpecType, sizeof...(Args)> expected_types { specTypeFor_<Args>()... };
1040
1041 if(parsed_specs_.size() != sizeof...(Args)){
1042 throw SpartaException("Sprintf format string \"")
1043 << format_str_ << "\" for " << getLocation() << " has " << parsed_specs_.size()
1044 << " specifier(s) but " << sizeof...(Args) << " argument(s) were given";
1045 }
1046
1047 for(std::size_t i = 0; i < parsed_specs_.size(); ++i){
1048 if(parsed_specs_[i] != expected_types[i]){
1049 throw SpartaException("Sprintf format string \"")
1050 << format_str_ << "\" for " << getLocation()
1051 << " has a type mismatch at specifier index " << i;
1052 }
1053 }
1054
1055 // One stream-to-ostringstream lambda per argument, invoked in order as specifiers are encountered
1056 const std::array<std::function<void(std::ostringstream&)>, sizeof...(Args)> streamers {
1057 [&args](std::ostringstream& o){ o << args; }...
1058 };
1059
1060 std::ostringstream oss;
1061 std::size_t arg_idx = 0;
1062 for(std::size_t i = 0; i < format_str_.size(); ++i){
1063 if(format_str_[i] != '%'){
1064 oss << format_str_[i];
1065 continue;
1066 }
1067
1068 const char spec = format_str_[++i];
1069 if(spec == '%'){
1070 oss << '%';
1071 }else{
1072 streamers[arg_idx++](oss);
1073 }
1074 }
1075
1076 NotificationSource<std::string>::postNotification(oss.str());
1077 }
1078
1079 private:
1080 enum class SpecType { STRING, INT };
1081
1082 // Maps an argument type to the specifier it must match (%i for non-bool integrals, %s otherwise)
1083 template <typename T>
1084 static constexpr SpecType specTypeFor_()
1085 {
1086 using PlainT = typename std::remove_cv<typename std::remove_reference<T>::type>::type;
1087 if constexpr (std::is_integral<PlainT>::value && !std::is_same<PlainT, bool>::value){
1088 return SpecType::INT;
1089 }else{
1090 return SpecType::STRING;
1091 }
1092 }
1093
1094 // Runtime scan of format_str for %s/%i/%% specifiers, in order of appearance
1095 static std::vector<SpecType> parseFormatSpecifiers_(const std::string& format_str)
1096 {
1097 std::vector<SpecType> specs;
1098 for(std::size_t i = 0; i < format_str.size(); ++i){
1099 if(format_str[i] != '%'){
1100 continue;
1101 }
1102
1103 if(i + 1 >= format_str.size()){
1104 throw SpartaException("Sprintf format string \"")
1105 << format_str << "\" ends with a lone '%' character";
1106 }
1107
1108 const char spec = format_str[++i];
1109 switch(spec){
1110 case 's':
1111 specs.push_back(SpecType::STRING);
1112 break;
1113 case 'i':
1114 specs.push_back(SpecType::INT);
1115 break;
1116 case '%':
1117 break; // Literal '%', not a specifier
1118 default:
1119 throw SpartaException("Sprintf format string \"")
1120 << format_str << "\" contains unsupported specifier '%" << spec
1121 << "'. Only %s and %i are supported";
1122 }
1123 }
1124 return specs;
1125 }

1233 void findMatchingSourcesBelowNode_(sparta::TreeNode& node,
1234 SrcList& bases)
1235 {
1236 auto* as_src = dynamic_cast<NotificationSourceBase*>(&node);
1237 if (as_src)
1238 {
1239 // If this MirrorNotificationSource happened to live under the search
1240 // of node, then we would certainly not want to add ourselves as a
1241 // concrete instance!
1242 if (as_src != this)
1243 {
1244 // We won't mirror anything that isn't explicitly allowed by the concrete
1245 // source.
1246 if (as_src->getNotificationID() == getNotificationID() &&
1247 as_src->getNotificationType() == getNotificationType())
1248 {
1249 bases.emplace_back(dynamic_cast<NotificationSourceBase*>(&node));
1250 }
1251 }
1252 }
1253

1255 {
1256 findMatchingSourcesBelowNode_(*n, bases);
